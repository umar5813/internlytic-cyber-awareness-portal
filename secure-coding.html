Vulnerable vs. Secure Code Examples

Vulnerable (SQL Injection Risk) 
// BAD: Direct string concatenation
const query = "SELECT * FROM users WHERE email = '" + userInput + "'";
db.execute(query);
Secure (Parameterized Query) 
// GOOD: Prepared statement separates code from data
const query = "SELECT * FROM users WHERE email = ?";
db.execute(query, [userInput]);
Vulnerable (Leaking Stack Traces) 
// BAD: Exposing database internals to client
catch (err) {
  res.status(500).send("DB Error: " + err.stack);
}
Secure (Generic Message + Internal Log) 
// GOOD: Log internally, show safe generic message
catch (err) {
  logger.error("Auth failure", { code: err.code });
  res.status(500).json({ error: "Something went wrong." });
}
